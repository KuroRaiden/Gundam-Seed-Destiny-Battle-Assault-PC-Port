cmake_minimum_required(VERSION 3.24)
project(GundamSeedRecomp LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8 /D_CRT_SECURE_NO_WARNINGS)
endif()
get_filename_component(WORK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../work" ABSOLUTE)
set(DEPS_DIR "${WORK_DIR}/deps" CACHE PATH "Checked out dependencies")
set(GENERATED_DIR "${WORK_DIR}/generated/us" CACHE PATH "Private generated US cartridge code")
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS_DIR}/SDL2" "${CMAKE_BINARY_DIR}/sdl" EXCLUDE_FROM_ALL)
set(SDL2_INCLUDE_DIR "${DEPS_DIR}/SDL2/include" CACHE PATH "" FORCE)
set(SDL2_LIBRARY SDL2-static CACHE STRING "" FORCE)
set(GBARECOMP_COMPILER_CACHE OFF CACHE STRING "" FORCE)
add_subdirectory("${DEPS_DIR}/gbarecomp" "${CMAKE_BINARY_DIR}/engine" EXCLUDE_FROM_ALL)
file(GLOB SHARDS CONFIGURE_DEPENDS "${GENERATED_DIR}/recompiled_*.cpp")
if(NOT SHARDS)
  message(FATAL_ERROR "Generate the US cartridge first with build.ps1")
endif()
add_library(seed_cartridge STATIC ${SHARDS}
  "${GENERATED_DIR}/dispatch_table.cpp" "${GENERATED_DIR}/symbol_map.cpp")
target_include_directories(seed_cartridge PRIVATE "${DEPS_DIR}/gbarecomp/src/armv4t" "${GENERATED_DIR}")
add_executable(GundamSeedRecomp src/main.cpp)
target_link_gbarecomp_runtime_stack(GundamSeedRecomp seed_cartridge)
add_library(seed_input STATIC src/input_state.cpp)
target_include_directories(seed_input PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_link_libraries(GundamSeedRecomp PRIVATE seed_input)
add_executable(seed_input_oracle tests/input_oracle.cpp)
target_link_libraries(seed_input_oracle PRIVATE seed_input gbarecomp_armv4t)
add_executable(seed_jp_asset_oracle tests/jp_asset_oracle.cpp)
target_link_libraries(seed_jp_asset_oracle PRIVATE gbarecomp_armv4t)
